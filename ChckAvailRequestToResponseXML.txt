    PROCEDURE RunCheckAvailability@1000000000(RequestText@1000000000 : Text) ResponseText : Text;

    LOCAL PROCEDURE ParseRequestXML@1000000002(XMLText@1000000000 : Text);
    BEGIN

      BEGIN
        // Load XML
        XmlDoc := XmlDoc.XmlDocument;
        XmlDoc.LoadXml(XMLText);


        NameTable := XmlDoc.NameTable;
        XmlNsMgr := XmlNsMgr.XmlNamespaceManager(NameTable);
      //Register XML namespaces for XPath queries
        XmlNsMgr.AddNamespace('d', 'http://example.com/supplierapi');
        XmlNsMgr.AddNamespace('h', 'http://example.com/supplierapi');
        XmlNsMgr.AddNamespace('i', 'http://example.com/supplierapi');

        Header.INIT;
      //Read header values
        XmlNode := XmlDoc.SelectSingleNode('//h:UUID', XmlNsMgr);
        IF NOT ISNULL(XmlNode) THEN
          Header."Session Token" := XmlNode.InnerText;

        XmlNode := XmlDoc.SelectSingleNode('//h:userName', XmlNsMgr);
        IF NOT ISNULL(XmlNode) THEN
          Header."User ID" := XmlNode.InnerText;

        XmlNode := XmlDoc.SelectSingleNode('//h:language', XmlNsMgr);
        IF NOT ISNULL(XmlNode) THEN
          Header."Language Code" := XmlNode.InnerText;
        //Insert Header record for request
        Header.INSERT(TRUE);


      //Retrieve item nodes for request
        XmlNodeList := XmlDoc.SelectNodes('//i:item', XmlNsMgr);

        FOR i := 0 TO XmlNodeList.Count - 1 DO BEGIN
          XmlNode := XmlNodeList.Item(i);

          Line.INIT;
          Line."Header Entry No." := Header."Entry No.";

          // Item Number
          IF NOT ISNULL(XmlNode.SelectSingleNode('i:itemNumber', XmlNsMgr)) THEN
            Line."Item No." :=
              XmlNode.SelectSingleNode('i:itemNumber', XmlNsMgr).InnerText;

          // UOM
          IF NOT ISNULL(XmlNode.SelectSingleNode('i:uom', XmlNsMgr)) THEN
            Line."Unit of Measure" :=
              XmlNode.SelectSingleNode('i:uom', XmlNsMgr).InnerText;

          // Description
          IF NOT ISNULL(XmlNode.SelectSingleNode('i:itemDescription', XmlNsMgr)) THEN
            Line.Description :=
              XmlNode.SelectSingleNode('i:itemDescription', XmlNsMgr).InnerText;

          // Quantity
          IF NOT ISNULL(XmlNode.SelectSingleNode('i:itemQuantity', XmlNsMgr)) THEN BEGIN
            QtyText := XmlNode.SelectSingleNode('i:itemQuantity', XmlNsMgr).InnerText;
            IF NOT EVALUATE(Line.Quantity, QtyText) THEN
              Line.Quantity := 0;
          END;

          Line.INSERT;
        END;
      END;
    END;

    LOCAL PROCEDURE ProcessLines@1000000003();
    BEGIN

      // Pricing and availability (business logic)
      BEGIN

        Line.RESET;
        Line.SETRANGE("Header Entry No.", Header."Entry No.");

        IF Line.FINDSET THEN
          REPEAT

            // Try to find item in NAV

      IF ItemRec.GET(Line."Item No.") THEN BEGIN

        // Pull item information from NAV
        Line.Description := ItemRec.Description;

        // Pricing
        Line."Unit Price" := ItemRec."Unit Price";

        // Extended price
        Line."Extended Price" :=
          Line.Quantity * Line."Unit Price";

        // Inventory (quantity available)
        ItemRec.CALCFIELDS(Inventory);
        Line."Available Quantity" := ItemRec.Inventory;

      Line."Item Exists" := TRUE;
      Line.Orderable := TRUE;
      Line."Orderable Reason" := '';
      //if item is blocked return orderable false due to item is blocked
      IF ItemRec.Blocked THEN BEGIN
          Line.Orderable := FALSE;
          Line."Orderable Reason" := 'Item is blocked';
      END;
      //if item is unavailable return orderable false due to out of stock
      IF ItemRec.Inventory <= 0 THEN BEGIN
          Line.Orderable := FALSE;
          Line."Orderable Reason" := 'Out of stock';
      END;

      //Use needs to order at least 1 item
        Line."Minimum Order Qty" := 1;

      END ELSE BEGIN
        //If item number is not found in system
        Line.Description := 'Item not found in NAV';
        Line."Unit Price" := 0;
        Line."Extended Price" := 0;
        Line."Available Quantity" := 0;

        Line."Item Exists" := FALSE;
        Line.Orderable := FALSE;
        Line."Orderable Reason" := 'Item does not exist';
        Line."Minimum Order Qty" := 0;

      END;
            Line.MODIFY;

          UNTIL Line.NEXT = 0;

      END;
    END;

    LOCAL PROCEDURE BuildResponseXML@1000000004() XMLText : Text;
    BEGIN


      BEGIN

        XmlDoc := XmlDoc.XmlDocument;

        Root := XmlDoc.CreateElement('CheckAvailabilityResponse');
        XmlDoc.AppendChild(Root);

        AddNode(Root,'Status','200');
        AddNode(Root,'Currency','USD');

        Line.RESET;
        Line.SETRANGE("Header Entry No.",Header."Entry No.");

        IF Line.FINDSET THEN
          REPEAT

            ItemEl := XmlDoc.CreateElement('Item');

            // Basic item info
            AddNode(ItemEl,'ItemNumber',Line."Item No.");
            AddNode(ItemEl,'UOM',Line."Unit of Measure");
            AddNode(ItemEl,'ItemDescription',Line.Description);

            // Availability
            AddNode(ItemEl,'ItemAvailability',
              DELCHR(FORMAT(Line."Available Quantity"),'=',','));

            // Pricing
            AddNode(ItemEl,'ItemListEach',
              DELCHR(FORMAT(Line."Unit Price"),'=',','));

            AddNode(ItemEl,'ItemNetEach',
              DELCHR(FORMAT(Line."Unit Price"),'=',','));

            AddNode(ItemEl,'ItemExtendedNet',
              DELCHR(FORMAT(Line."Extended Price"),'=',','));

            // Requested qty
            AddNode(ItemEl,'ItemQuantity',
              DELCHR(FORMAT(Line.Quantity),'=',','));

            // Minimum order
            AddNode(ItemEl,'MinimumOrderQuantity',
              DELCHR(FORMAT(Line."Minimum Order Qty"),'=',','));

            // Exists
            IF Line."Item Exists" THEN
              AddNode(ItemEl,'ItemExists','true')
            ELSE
              AddNode(ItemEl,'ItemExists','false');

            // Orderable
            IF Line.Orderable THEN
              AddNode(ItemEl,'Orderable','true')
            ELSE
              AddNode(ItemEl,'Orderable','false');

            AddNode(ItemEl,
                    'OrderableReason',
                    Line."Orderable Reason");
            Root.AppendChild(ItemEl);

          UNTIL Line.NEXT = 0;

        XMLText := XmlDoc.OuterXml;

      END;
    END;
